```bash

  --revision 7ae557604adf67be50417f59c2c2f167def9a775
uv run --locked python scripts/prepare_model.py check \
  --revision 7ae557604adf67be50417f59c2c2f167def9a775

  --revision 7ae557604adf67be50417f59c2c2f167def9a775 --gpu
```
